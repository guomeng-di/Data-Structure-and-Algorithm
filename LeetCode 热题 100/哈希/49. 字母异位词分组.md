# 49. 字母异位词分组

# 初次尝试:
超出时间限制
```c++
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        vector<vector<string>> ans;
        vector<bool> status(strs.size(),0);
        //set<string> st;
        int q=0;
        for(int i=0;i<strs.size();i++){
            if(status[i]) continue;
            status[i]=1;
            string c=strs[i];
            ans.emplace_back();
            ans.back().push_back(strs[i]);
            sort(c.begin(),c.end());
            // st.insert(c);
            for(int j=i+1;j<strs.size();j++){
                if(status[j]) continue;
                string d=strs[j];
                sort(d.begin(),d.end());
                if(d==c){ 
                  ans.back().push_back(strs[j]);
                  status[j]=1;
                }else{
                    continue;
                }
            }
            q++;
        }
        return ans;
    }
};
```
# 优化:
## 当前代码思想:
暴力双重循环：
用status[]标记已经分到组里的字符串   
i 遍历每一个没被分组的字符串，把它当做本组基准，排序得到 key    
j 从 i+1 往后扫描，把排序后等于 key 的全部归到同一组，打上 status 标记   

## 优化思路 1： 
## 预处理，提前把所有字符串的排序 key 算好存起来  
只对每个字符串排序1 次，不再 j 循环里反复 sort。
开辟一个vector<string>> keys，预先算出每个 str 对应的排序 key。
之后 i,j 循环直接拿预计算好的 key 对比，不再重复 sort。

```c++
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        vector<vector<string>> ans;
        vector<bool> status(strs.size(),0);
        vector<string> keys(strs.size());
        for(int i=0;i<strs.size();i++){
            keys[i]=strs[i];
            sort(keys[i].begin(), keys[i].end());
        }
        for(int i=0;i<strs.size();i++){
            if(status[i]) continue;
            status[i]=1;
            //string c=strs[i];
            ans.emplace_back();
            ans.back().push_back(strs[i]);
            string target = keys[i];
            //sort(c.begin(),c.end());
            // st.insert(c);
            for(int j=i+1;j<strs.size();j++){
                if(status[j]) continue;
                //string d=strs[j];
                //sort(d.begin(),d.end());
                string curr= keys[j];
                if(curr==target){ 
                  ans.back().push_back(strs[j]);
                  status[j]=1;
                }else{
                    continue;
                }
            }
            
        }
        return ans;
    }
};
```

## 优化思路 2:
### 核心思想：排序后的字符串作为key，value存一组原字符串，一次遍历完成分组

```c++
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        vector<vector<string>> ans;
        unordered_map<string, vector<string>> mp;
        for(int i=0;i<strs.size();i++){
            string key=strs[i];
            sort(key.begin(),key.end());
            mp[key].push_back(strs[i]);
        }

        for(auto entry:mp){
            ans.push_back(entry.second);
        }
        return ans;
    }
};
```
