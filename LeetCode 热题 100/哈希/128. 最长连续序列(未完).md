# 128. 最长连续序列
## 初次尝试:  
使用vector去重方法:  
sort->unique->erase  
```c++
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        if(nums.size()==0) return 0;
        sort(nums.begin(),nums.end());
        auto last=unique(nums.begin(),nums.end());
        nums.erase(last,nums.end());
        int ans=1;int cnt=1;
        for(int i=1;i<nums.size();i++){
            if(nums[i]-nums[i-1]==1){
                cnt++;
            }else{
                ans=max(ans,cnt);
                cnt=1;
            }
        }
        return max(ans,cnt);
    }
};
```
