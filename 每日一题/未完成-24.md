# 3488. 距离最小相等元素查询
## 暴力:第614个超时
### 算法:二分指针
```c++
class Solution {
public:
    vector<int> solveQueries(vector<int>& nums, vector<int>& queries) {
        if(nums.size()==0){
            vector<int> ans(queries.size(),-1);
            return ans;
        }

        vector<int> ans(queries.size(),-1);
        for(int i=0;i<queries.size();i++){
            int target=nums[queries[i]],start=queries[i];
            int left=(start-1+nums.size())%nums.size(),right=(start+1+nums.size())%nums.size();
            while(left!=right){
                if(target==nums[left]){
                    int q1=abs(start-left);
                    ans[i]=min(q1,(int)(nums.size()-q1));
                    break;
                }
                else if(target==nums[right]){
                    int q2=abs(start-right);
                    ans[i]=min(q2,(int)(nums.size()-q2));
                    break;
                }
                else{
                    left=(left-1+nums.size())%nums.size();
                    right=(right+1+nums.size())%nums.size();
                }
            }
            //此时有可能都为答案,但因left==right退出
            if(target==nums[left]&&start!=left){
                    int q1=abs(start-left);
                    ans[i]=min(q1,(int)(nums.size()-q1));
                }
        }
        return ans;
    }
};
```
