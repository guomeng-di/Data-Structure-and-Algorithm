# 1871. 跳跃游戏 VII

##   初始思路:

遍历string s,找到'0'就看步数是否满足要求的关系

```c++
class Solution {
public:
    bool canReach(string s, int minJump, int maxJump) {
        int i=0;
        for(int j=1;j<s.size();j++){
            if(s[j]=='0'){
                int a=min(i+maxJump,(int)s.size()-1);
                if(!(i+minJump<=j&&(j<=a))){
                    return 0;
                }
                i=j;
            }
        }
        return 1;
    }
};
```

##   问题:

s ="0000000000"

minJump =2

maxJump =5

此时,我们会因为step=1小于minJump而输出fause

然而,是可以跳步的

##   改进:

1.采用可跳范围内寻找'0'的方法

2.遇到第一个满足条件的就跳的"贪心"思想时不对的,正确方法是bfs或dp

```c++
class Solution {
public:
    bool canReach(string s, int minJump, int maxJump) {
        int len=s.size();//长度
        if(s[len-1]=='1') return 0;//最后一个字母是1->不符合要求

        vector<bool> st(len,0);//在高皇太数组->每个数能否到达,是否被访问过
        queue<int> q;//bfs队列
        q.push(0);//起点入队
        st[0]=1;

        int last=0;//上一次可以访问到的最右端

        while(!q.empty()){
            int curr=q.front();//取出对首,后续看它能访问的区域
            q.pop();//弹出对首

            int l=curr+minJump,r=curr+maxJump;
            l=max(l,last+1),r=min(r,len-1);

            for(int i=l;i<=r;i++){
                if(s[i]=='0'&&st[i]==0){
                    if(i==len-1) return 1;

                    st[i]=1;
                    q.push(i);
                }
            }
            last=r;//更新本轮最右端
        }
        return 0;// 队列空了还没找到终点，不可达
    }
};
```
