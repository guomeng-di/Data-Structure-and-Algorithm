# 考察**计数排序**
```c++
class Solution {
public:
    int maxIceCream(vector<int>& costs, int coins) {
        if(costs.empty()) return 0;
        int max_val=*max_element(costs.begin(),costs.end());
        vector<int> paixu(max_val+1,0);
        for(int i=0;i<costs.size();i++){
            paixu[costs[i]]++;
            //paixu[1]=2,paixu[2]=1,paixu[3]=1,paixu[4]=1
            //paixu[x]=n(x出现了n次)
        }
        //更新数组(实现排序)
        int idex=0;
        for(int i=0;i<=max_val;i++){
            while(paixu[i]--) costs[idex++]=i;
        }
// //
// for(int i=0;i<costs.size();i++){
//     cout<<costs[i]<<endl;
// }
        //求结果
        int ans=0;
        for(int i=0;i<costs.size();i++){
            if(coins-costs[i]>=0){
                ans++;
                coins-=costs[i];
            }
            else return ans;
        }
        return ans;
    }
};
//计数排序
```
