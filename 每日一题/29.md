# to_string函数实现了整数转字符串
# 3726. 移除十进制表示中的所有零
class Solution {
public:
    long long removeZeros(long long n) {
        string s=to_string(n);
        long long ans=0;
        for(int i=0;i<s.size();i++){
            if(s[i]!='0'){
                ans=ans*10+(s[i]-'0');
            }
        }
        return ans;
    }
};
