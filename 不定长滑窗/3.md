# 不定长滑窗求子数组个数
## 思路核心：
固定右端点 right，找到最小合法左边界 left，此时以 right 结尾的合法子数组数量 = right-left+1  
(这个区间内所有以 right 结尾的子数组全部合法：[right]、[right-1,right]、...、[left,right])  
```c++
class Solution {
public:
    int beautifulBouquet(vector<int>& flowers, int cnt) {
        unordered_map<int,int> mp;//存每个数出现的次数
        int left=0,ans=0;//左端点
        for(int right=0;right<flowers.size();right++){
            int x=flowers[right];//品种
            mp[x]++;//x品种出现次数
            while(mp[x]>cnt){
                mp[flowers[left]]--;//左端点右移
                left++;
            }
            ans+=right-left+1;
        }
        return ans;
    }
};
