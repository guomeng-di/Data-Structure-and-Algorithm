```c++

class FrontMiddleBackQueue {
    vector<int> arr;
public:
    FrontMiddleBackQueue() {
        
    }
    
    void pushFront(int val) {
        arr.insert(arr.begin(),val);
    }
    
    void pushMiddle(int val) {
        arr.insert(arr.begin()+(arr.size()/2),val);
    }
    
    void pushBack(int val) {
        arr.push_back(val);
    }
    
    int popFront() {
        if(arr.size()==0) return -1;
        int ans=arr[0];
        arr.erase(arr.begin());
        return ans;
    }
    
    int popMiddle() {
        if(arr.size()==0) return -1;
        int ans=arr[(arr.size()-1)/2];
        arr.erase(arr.begin()+(arr.size()-1)/2);
        return ans;
    }
    
    int popBack() {
        if(arr.size()==0) return -1;
        int ans=arr[arr.size()-1];
        arr.pop_back();
        return ans;
    }
};
